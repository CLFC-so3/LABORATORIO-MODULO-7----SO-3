sudo nano /etc/samba/smb.conf
(dentro de sudo nano /etc/samba/smb.conf, agregar al fondo)
[compartir]
   path = /samba/compartir
   browseable = yes
   writeable = yes
   valid users = zamba
   create mask = 0777
   directory mask = 0777
   guest ok = yes


sudo systemctl restart smb nmb
